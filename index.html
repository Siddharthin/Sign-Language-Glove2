#include <WiFi.h> #include <WebSocketsServer.h> #include <ArduinoJson.h> #include <Wire.h> #include <MPU6050.h> #include <SoftwareSerial.h> // WiFi credentials const char* ssid = "YOUR_WIFI_SSID"; const char* password = "YOUR_WIFI_PASSWORD"; // WebSocket server WebSocketsServer webSocket = WebSocketsServer(81); // MPU6050 sensor MPU6050 mpu; bool mpuConnected = false; // GPS module (using SoftwareSerial) SoftwareSerial gpsSerial(16, 17); // RX, TX pins bool gpsConnected = false; // Flex sensors (analog pins) const int flexPins[4] = {A0, A1, A2, A3}; bool flexConnected = false; // Buzzer pin const int buzzerPin = 2; bool buzzerState = false; // Data collection - Optimized for faster transmission unsigned long lastSensorRead = 0; const unsigned long sensorInterval = 50; // 50ms = 20Hz for faster updates void setup() { Serial.begin(115200); // Initialize pins pinMode(buzzerPin, OUTPUT); digitalWrite(buzzerPin, LOW); // Initialize I2C for MPU6050 Wire.begin(); // Try to initialize MPU6050 mpu.initialize(); mpuConnected = mpu.testConnection(); if (mpuConnected) { Serial.println("MPU6050 connected successfully"); } else { Serial.println("MPU6050 connection failed"); } // Initialize GPS gpsSerial.begin(9600); gpsConnected = true; // Assume GPS is connected, will validate with data // Test flex sensors testFlexSensors(); // Connect to WiFi WiFi.begin(ssid, password); while (WiFi.status() != WL_CONNECTED) { delay(1000); Serial.println("Connecting to WiFi..."); } Serial.println("WiFi connected!"); Serial.print("IP address: "); Serial.println(WiFi.localIP()); // Start WebSocket server webSocket.begin(); webSocket.onEvent(webSocketEvent); Serial.println("WebSocket server started on port 81"); } void loop() { webSocket.loop(); // Read sensors at regular intervals if (millis() - lastSensorRead >= sensorInterval) { sendSensorData(); lastSensorRead = millis(); } // Reduced delay for faster response delay(1); } void testFlexSensors() { // Test if flex sensors are connected by reading values int validReadings = 0; for (int i = 0; i < 4; i++) { int reading = analogRead(flexPins[i]); if (reading > 10 && reading < 4090) { // Valid range validReadings++; } } flexConnected = (validReadings >= 2); // At least 2 sensors connected if (flexConnected) { Serial.println("Flex sensors detected"); } else { Serial.println("No flex sensors detected"); } } void sendSensorData() { // Use smaller JSON buffer for faster processing DynamicJsonDocument doc(512); // MPU6050 data - only send if connected if (mpuConnected) { JsonObject mpuObj = doc.createNestedObject("mpu6050"); int16_t ax, ay, az, gx, gy, gz; mpu.getMotion6(&ax, &ay, &az, &gx, &gy, &gz); // Convert to g and degrees/second with reduced precision for speed mpuObj["connected"] = true; mpuObj["accel_x"] = round((ax / 16384.0) * 100) / 100.0; // 2 decimal places mpuObj["accel_y"] = round((ay / 16384.0) * 100) / 100.0; mpuObj["accel_z"] = round((az / 16384.0) * 100) / 100.0; mpuObj["gyro_x"] = round((gx / 131.0) * 10) / 10.0; // 1 decimal place mpuObj["gyro_y"] = round((gy / 131.0) * 10) / 10.0; mpuObj["gyro_z"] = round((gz / 131.0) * 10) / 10.0; // Calculate pitch and roll with reduced precision float accel_x = ax / 16384.0; float accel_y = ay / 16384.0; float accel_z = az / 16384.0; mpuObj["pitch"] = round(atan2(accel_y, sqrt(accel_x * accel_x + accel_z * accel_z)) * 180 / PI * 10) / 10.0; mpuObj["roll"] = round(atan2(-accel_x, accel_z) * 180 / PI * 10) / 10.0; } // Flex sensor data - only send if connected if (flexConnected) { JsonObject flexObj = doc.createNestedObject("flex"); flexObj["connected"] = true; flexObj["avg1"] = analogRead(flexPins[0]); flexObj["avg2"] = analogRead(flexPins[1]); flexObj["avg3"] = analogRead(flexPins[2]); flexObj["avg4"] = analogRead(flexPins[3]); } // Skip GPS for now to reduce packet size and improve speed // GPS can be added back if needed with separate transmission // Send JSON data with minimal serialization String jsonString; serializeJson(doc, jsonString); webSocket.broadcastTXT(jsonString); } void webSocketEvent(uint8_t num, WStype_t type, uint8_t * payload, size_t length) { switch(type) { case WStype_DISCONNECTED: Serial.printf("[%u] Disconnected!\n", num); break; case WStype_CONNECTED: { IPAddress ip = webSocket.remoteIP(num); Serial.printf("[%u] Connected from %d.%d.%d.%d\n", num, ip[0], ip[1], ip[2], ip[3]); break; } case WStype_TEXT: { String message = String((char*)payload); Serial.printf("[%u] Received: %s\n", num, message.c_str()); // Handle commands if (message == "BUZZER_ON") { digitalWrite(buzzerPin, HIGH); buzzerState = true; webSocket.sendTXT(num, "Buzzer ON"); } else if (message == "BUZZER_OFF") { digitalWrite(buzzerPin, LOW); buzzerState = false; webSocket.sendTXT(num, "Buzzer OFF"); } else if (message == "REQUEST_DATA") { sendSensorData(); } else if (message == "CALIBRATE") { // Calibrate sensors if (mpuConnected) { mpu.setXAccelOffset(0); mpu.setYAccelOffset(0); mpu.setZAccelOffset(0); } webSocket.sendTXT(num, "Calibration complete"); } else if (message == "RESET") { webSocket.sendTXT(num, "Resetting ESP32..."); delay(1000); ESP.restart(); } else if (message == "STATUS") { DynamicJsonDocument statusDoc(512); statusDoc["mpu6050"] = mpuConnected; statusDoc["flex"] = flexConnected; statusDoc["gps"] = gpsConnected; statusDoc["buzzer"] = buzzerState; statusDoc["wifi_rssi"] = WiFi.RSSI(); statusDoc["free_heap"] = ESP.getFreeHeap(); String statusString; serializeJson(statusDoc, statusString); webSocket.sendTXT(num, statusString); } break; } default: break; } }

Training new model...") self.train_initial_model() def save_model(self): """Save trained model and scaler""" try: with open('gesture_model.pkl', 'wb') as f: pickle.dump(self.model, f) with logger.error(f"Error training initial model: {e}") def extract_features(self, sensor_data): """Extract features from sensor data""" try: features = [] # Flex sensor features (4 values) if 'flex' in sensor_data['mpu6050'].get('gyro_y', 0), sensor_data['mpu6050'].get('gyro_z', 0) ] features.extend(imu_data) else: features.extend([0, 0, 0, 0, 0, 0]) # Orientation features (2 values) if 'mpu6050' in sensor_data and sensor_data['mpu6050'].get('connected'): orientation_data = [ sensor_data['mpu6050'].get('pitch', 0), sensor_data['mpu6050'].get('roll', 0) ] features.extend(orientation_data) else: features.extend([0, 0]) return np.array(features).reshape(1, -1) except Exception as e: logger.error(f"Error extracting features: {e}") return np.zeros((1, 12)) def predict_gesture(self, sensor_data):
